# === CLAUDE CODE AUTHENTICATION =============================================
# Uncomment ONE section below matching your setup. Claude Code checks
# credentials in priority order — if multiple are set, the highest-priority
# one wins. See: https://code.claude.com/docs/en/authentication
#
# Priority order:
#   1. Bedrock / Vertex / Foundry (cloud provider)
#   2. ANTHROPIC_AUTH_TOKEN (proxy / gateway — used by Options C and F)
#   3. ANTHROPIC_API_KEY (direct API — pay-per-use)
#   4. CLAUDE_CODE_OAUTH_TOKEN (subscription — Max, Pro, Team, Enterprise)
#   5. Interactive login (browser-based, no env var needed)
# ============================================================================

# --- Option A: Anthropic API Key (pay-per-use) ---
# Get a key: https://console.anthropic.com/
# ANTHROPIC_API_KEY=sk-ant-api03-your_key_here

# --- Option B: Max / Pro / Team / Enterprise Subscription ---
# Run `claude setup-token` once inside the container to generate a long-lived
# OAuth token (valid ~1 year), then paste it here for persistent auth.
# NOTE: Do NOT also set ANTHROPIC_API_KEY — it takes priority and you'll be
# billed per-use instead of through your subscription.
# CLAUDE_CODE_OAUTH_TOKEN=your_oauth_token_here

# --- Option C: OpenRouter (third-party model gateway) ---
# Docs: https://openrouter.ai/docs/guides/coding-agents/claude-code-integration
# Get a key: https://openrouter.ai/settings/keys
# If you previously logged in with Anthropic, run /logout in Claude Code first.
# Leave the ANTHROPIC_API_KEY blank below
# ANTHROPIC_BASE_URL=https://openrouter.ai/api
# ANTHROPIC_AUTH_TOKEN=your_openrouter_api_key_here
# ANTHROPIC_API_KEY=
# OPENROUTER_API_KEY=your_openrouter_api_key_here

# --- Option C, extended: OpenAI GPT models through OpenRouter ---
# OpenRouter also serves OpenAI GPT models, and DAAF runs its full agentic
# stack on them (multi-tool loops, subagent dispatch, two-tier routing) with
# no proxy, no rebuild — just the Option C lines above plus these
# (validated live 2026-07-09):
#
# ANTHROPIC_MODEL=openai/gpt-5.6-sol                # your session model
# ANTHROPIC_DEFAULT_OPUS_MODEL=openai/gpt-5.6-sol   # strong tier / Opus-analog (1,050,000-token window)
# ANTHROPIC_DEFAULT_SONNET_MODEL=openai/gpt-5.6-terra # fast tier / Sonnet-analog (1,050,000-token window)
# CLAUDE_CODE_MAX_CONTEXT_TOKENS=1050000            # match your SESSION model's real window
#
# Notes: context-utilization numbers are estimates on GPT sessions (OpenRouter
# doesn't serve Claude Code's token-counting endpoint), and GPT models keep
# DAAF's conservative context-quality thresholds by design. The GPT-5.6 family
# uses OpenAI's tier-analog naming -- Sol (strong/Opus-analog), Terra
# (mid/Sonnet-analog), Luna (economy/Haiku-analog) -- all with 1,050,000-token
# windows; older gpt-5.5/5.4/5.2 slugs still work. See the installation guide's
# "GPT (OpenAI) models via OpenRouter" section for current model guidance.

# --- Optional: Subagent model routing (applies to Options C and F) ---------
# DAAF routes specialist subagents to Claude model tiers: 'opus' for high-
# judgment work (planning, QA review, verification, debugging) and 'sonnet'
# for well-specified work (fetching, profiling, notebook assembly).
#
# If you point Claude Code at an alternative provider (e.g., OpenRouter
# serving GPT, GLM, Kimi, Qwen — or the Option F shim below), those Claude
# aliases don't exist on your endpoint. Remap them so DAAF's two-tier routing
# uses YOUR models instead:
#
# ANTHROPIC_DEFAULT_OPUS_MODEL=your-strong-model-slug      # e.g. openai/gpt-5.6-sol or z-ai/glm-5.2
# ANTHROPIC_DEFAULT_SONNET_MODEL=your-fast-model-slug      # e.g. openai/gpt-5.6-terra or z-ai/glm-5.2-air
#
# Or, to switch tiered routing off and run every subagent on a single model:
#
# CLAUDE_CODE_SUBAGENT_MODEL=your-model-slug
#
# If you set none of these on a non-Claude session, DAAF's model-routing guard
# blocks mismatched Claude-tier dispatches and subagents run on your session
# model.

# --- Option D: Amazon Bedrock ---
# Docs: https://code.claude.com/docs/en/bedrock-vertex
# CLAUDE_CODE_USE_BEDROCK=1
# AWS_REGION=us-east-1
# ANTHROPIC_BEDROCK_BASE_URL=https://bedrock-runtime.us-east-1.amazonaws.com

# --- Option E: Google Vertex AI ---
# Docs: https://code.claude.com/docs/en/bedrock-vertex
# CLAUDE_CODE_USE_VERTEX=1
# ANTHROPIC_VERTEX_PROJECT_ID=your_project_id
# CLOUD_ML_REGION=us-central1

# --- Option F: OpenAI API directly (DAAF provider shim) ---
# Runs GPT models against your own OpenAI API key, with no gateway in between.
# OpenAI's API doesn't speak Anthropic's wire format, so DAAF ships a small
# translation shim (scripts/provider_shim/ — one auditable file, zero extra
# dependencies). When DAAF_PROVIDER_SHIM is set, the container starts and
# supervises the shim automatically at boot; there is nothing to run manually.
# Get a key: https://platform.openai.com/api-keys
# Setup guide: user_reference/01_installation_and_quickstart.md, Option F
#
# NOTE: Requires ONE image rebuild after first enabling this option (the
# auto-launch hook is baked into the container entrypoint):
#   bash rebuild_daaf.sh          (macOS / Linux)
#   .\rebuild_daaf.ps1            (Windows)
# After that, config changes here only need the usual container recreate.
#
# DAAF_PROVIDER_SHIM=openai
# OPENAI_API_KEY=sk-your_openai_key_here
# ANTHROPIC_BASE_URL=http://127.0.0.1:4141
# ANTHROPIC_AUTH_TOKEN=daaf-shim-local
# ANTHROPIC_API_KEY=
#
# Model selection — bare OpenAI slugs here (no "openai/" prefix):
# ANTHROPIC_MODEL=gpt-5.6-sol
# ANTHROPIC_DEFAULT_OPUS_MODEL=gpt-5.6-sol
# ANTHROPIC_DEFAULT_SONNET_MODEL=gpt-5.6-terra
# CLAUDE_CODE_MAX_CONTEXT_TOKENS=1050000            # gpt-5.6 family (sol/terra/luna) + gpt-5.5/5.4: 1050000; gpt-5.2: 400000
#
# Advanced shim tuning (defaults are correct for OpenAI; rarely needed):
# SHIM_PORT=4141
# SHIM_BACKEND_BASE_URL=https://api.openai.com/v1
# SHIM_BACKEND_API_KEY=                             # defaults to OPENAI_API_KEY
# SHIM_STRIP_MODEL_PREFIX=openai/                   # literal prefix stripped from model ids before the backend; only needed with prefixed slugs — leave unset for the bare slugs shown above
#
# Troubleshooting (inside the container):
#   bash /daaf/scripts/provider_shim/start_shim.sh --status
#   cat /daaf/scripts/provider_shim/logs/shim.log

# === MULTI-INSTANCE SETTINGS (OPTIONAL) =====================================
# Run more than one DAAF install on the same machine by giving each install
# folder its own project name and host ports. All DAAF scripts read these
# values automatically. Leave everything unset for a standard single install —
# the defaults match the original behavior exactly.
#
# NOTE: If you run `docker compose` commands directly (outside the provided
# scripts), also create a .env file in this folder with the same values —
# bare docker compose does not read environment_settings.txt for these.
# After changing any of these on an existing install, recreate the container:
#   docker compose down, then bash run_daaf.sh (or .\run_daaf.ps1)
# ============================================================================

# Compose project name (prefixes the container and data volume). Default: daaf
# WARNING: changing this on an EXISTING install detaches it from its current
# data volume — only set this when creating a NEW second install.
# DAAF_PROJECT_NAME=daaf2

# Host ports for the web services. Defaults: 2718 / 2719 / 2720
# DAAF_PORT_MARIMO=3718
# DAAF_PORT_LOGVIEWER=3719
# DAAF_PORT_VSCODE=3720

# ----------------------------------------------------------------------------
# Developer tooling (framework developers only)
# ----------------------------------------------------------------------------
# DAAF_DEV is an OPT-IN, BUILD-TIME flag. Set it to 1 to have the Docker image
# additionally install the in-container test toolchain (shellcheck, bats,
# PowerShell 7, Pester, PSScriptAnalyzer) so you can run DAAF's own shell and
# PowerShell test suites inside the container and reproduce what CI runs:
#
#   bats tests/bash/
#   pwsh -NoProfile -Command "Invoke-Pester -Path ./tests/powershell/"
#
# This is a BUILD flag, not a runtime setting: it only takes effect at
# `docker compose build`, so after changing it you must rebuild
# (rebuild_daaf.sh / rebuild_daaf.ps1). When unset or 0 (the default for every
# normal install), NONE of this tooling is installed and the image is identical
# to a standard build -- leaving it off costs you nothing.
#
# DAAF_DEV=1
